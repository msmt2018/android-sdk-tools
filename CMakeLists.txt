cmake_minimum_required(VERSION 3.19)

# some third-party projects still declare cmake_minimum_required < 3.5,
# which is rejected by CMake 4.x unless a policy minimum is provided
if(NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "")
endif()

project(sdk-tools C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

list(PREPEND CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/cmake/modules)

# platform tools version, read from patches/misc/platform_tools_version.h
file(STRINGS ${PROJECT_SOURCE_DIR}/patches/misc/platform_tools_version.h _tools_version_line
    REGEX "^#define[ \t]+PLATFORM_TOOLS_VERSION[ \t]+\"[^\"]+\"")
string(REGEX REPLACE "^.*\"([^\"]+)\".*$" "\\1" TOOLS_VERSION "${_tools_version_line}")
if(NOT TOOLS_VERSION)
    message(FATAL_ERROR "can't read PLATFORM_TOOLS_VERSION from patches/misc/platform_tools_version.h")
endif()

set(SRC ${PROJECT_SOURCE_DIR}/src)

# 64-bit off_t for lseek.
add_definitions(-D_FILE_OFFSET_BITS=64)

# bundled zlib must exist before any find_package(ZLIB) of the third-party projects
include(cmake/zlib.cmake)
# toolchain independent settings (static link, system libraries ...)
include(cmake/platform.cmake)

message(STATUS "sdk-tools version: ${TOOLS_VERSION}")
message(STATUS "target: ${SDK_TARGET_OS}-${CMAKE_SYSTEM_PROCESSOR}, compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
message(STATUS "static link: ${SDK_STATIC}, strip: ${SDK_STRIP}")

# options of the third-party projects
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)

# SDK executables are linked statically for Android/Termux distribution.
# Pin PCRE2's 8-bit library and Expat to static targets; otherwise CMake can
# emit a shared libexpat.so into a -static link and leave -lpcre2-8 unresolved.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)

set(PROTOC_COMPILER)

if(DEFINED PROTOC_PATH)
    set(PROTOC_COMPILER ${PROTOC_PATH})
    
    if(NOT EXISTS ${PROTOC_COMPILER})
        unset(PROTOC_PATH CACHE)
        message(FATAL_ERROR "Invalid protoc: ${PROTOC_COMPILER}, please check if the path is correct")
    endif()
    
    # check protoc version
    execute_process(
        COMMAND ${PROTOC_COMPILER} --version
        OUTPUT_VARIABLE PTOTOBUF_VERSION
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    )
    
    if(NOT PTOTOBUF_VERSION MATCHES "^libprotoc*")
        unset(PROTOC_PATH CACHE)
        message(FATAL_ERROR "Invalid protoc: ${PROTOC_COMPILER}, this may not be an executable")
    endif()
    
    string(REGEX MATCH "([0-9]+)\\.([0-9]+)\\.([0-9]+)" VERSIONS ${PTOTOBUF_VERSION})
    # please note that if the protobuf version is too high, you may encounter the following error
    # error: This file was generated by a newer version of protoc 
    if(VERSIONS VERSION_GREATER "3.21.12")
        message(WARNING "${PTOTOBUF_VERSION}, the protobuf version is too high, may be incompatible!!")
    endif()
endif()

# thrid-party libraries
# BoringSSL in Platform Tools 34.x keeps its generated CMake project under src/.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/boringssl/CMakeLists.txt")
    add_subdirectory(src/boringssl EXCLUDE_FROM_ALL)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/boringssl/src/CMakeLists.txt")
    add_subdirectory(src/boringssl/src EXCLUDE_FROM_ALL)
else()
    message(FATAL_ERROR "BoringSSL CMakeLists.txt not found in the checked-out AOSP source")
endif()
add_subdirectory(src/fmtlib EXCLUDE_FROM_ALL)
# Older Platform Tools tags do not always ship standalone CMake projects for
# these optional compression/regex sources. Add only projects that provide
# Older Platform Tools AOSP PCRE2 snapshots keep the maintained CMake
# project under dist2/, not at the repository root. Build the pinned source
# checkout's real 8-bit library because AAPT links against pcre2-8.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/pcre/dist2/CMakeLists.txt")
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static third-party libraries" FORCE)
    set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
    set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
    set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
    set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
    set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(src/pcre/dist2 EXCLUDE_FROM_ALL)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/pcre/CMakeLists.txt")
    add_subdirectory(src/pcre EXCLUDE_FROM_ALL)
else()
    message(FATAL_ERROR "PCRE2 CMakeLists.txt not found in the checked-out AOSP source")
endif()
if(NOT TARGET pcre2-8)
    message(FATAL_ERROR "The pinned AOSP PCRE2 source did not define the required pcre2-8 target")
endif()
add_subdirectory(src/libpng EXCLUDE_FROM_ALL)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/zopfli/CMakeLists.txt")
    add_subdirectory(src/zopfli EXCLUDE_FROM_ALL)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/zopfli/src/zopfli/zopfli_lib.c")
    # Older pinned AOSP Zopfli snapshots have no CMakeLists.txt. Build the
    # upstream library sources directly instead of passing -llibzopfli to ld.
    file(GLOB ZOPFLI_LIBRARY_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/zopfli/src/zopfli/*.c")
    add_library(libzopfli STATIC ${ZOPFLI_LIBRARY_SOURCES})
    set_target_properties(libzopfli PROPERTIES OUTPUT_NAME zopfli)
    target_include_directories(libzopfli PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/src/zopfli/src/zopfli")
    if(UNIX)
        target_link_libraries(libzopfli PUBLIC m)
    endif()
else()
    message(FATAL_ERROR "Pinned Zopfli source has neither CMakeLists.txt nor zopfli_lib.c")
endif()
add_subdirectory(src/jsoncpp EXCLUDE_FROM_ALL)
# AOSP's vendored JsonCpp CMake revisions expose jsoncpp_lib rather than the
# legacy jsoncpp_static target consumed by build-tools/aapt.cmake.
if(NOT TARGET jsoncpp_static AND TARGET jsoncpp_lib)
    add_library(jsoncpp_static ALIAS jsoncpp_lib)
endif()
if(NOT TARGET jsoncpp_static)
    message(FATAL_ERROR "JsonCpp was added but did not define jsoncpp_static or jsoncpp_lib")
endif()
# This AOSP Expat snapshot uses BUILD_shared (not EXPAT_SHARED_LIBS).
# Force its actual CMake option to STATIC so AAPT can be linked fully static.
set(BUILD_shared OFF CACHE BOOL "Build a static Expat library" FORCE)
set(BUILD_tools OFF CACHE BOOL "" FORCE)
set(BUILD_examples OFF CACHE BOOL "" FORCE)
set(BUILD_tests OFF CACHE BOOL "" FORCE)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/expat/expat/CMakeLists.txt")
    add_subdirectory(src/expat/expat EXCLUDE_FROM_ALL)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/expat/CMakeLists.txt")
    add_subdirectory(src/expat EXCLUDE_FROM_ALL)
else()
    message(FATAL_ERROR "Expat CMakeLists.txt not found in the checked-out AOSP source")
endif()
get_target_property(EXPAT_TARGET_TYPE expat TYPE)
if(NOT EXPAT_TARGET_TYPE STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Expected the pinned AOSP Expat target to be static, got: ${EXPAT_TARGET_TYPE}")
endif()
add_subdirectory(src/zstd/build/cmake EXCLUDE_FROM_ALL)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/lz4/build/cmake/CMakeLists.txt")
    add_subdirectory(src/lz4/build/cmake EXCLUDE_FROM_ALL)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/lz4/CMakeLists.txt")
    add_subdirectory(src/lz4 EXCLUDE_FROM_ALL)
else()
    message(STATUS "LZ4: no standalone CMake project in this AOSP tag; skipping")
endif()
# The pinned LZ4 revisions do not consistently export the legacy target name
# referenced by sload_f2fs. Supply a static implementation from the same source
# checkout when the upstream CMake project omits it.
if(NOT TARGET lz4_static AND TARGET lz4)
    get_target_property(_lz4_type lz4 TYPE)
    if(_lz4_type STREQUAL "STATIC_LIBRARY")
        add_library(lz4_static ALIAS lz4)
    endif()
endif()
if(NOT TARGET lz4_static AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/lz4/lib/lz4.c")
    file(GLOB LZ4_STATIC_SOURCES CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/src/lz4/lib/*.c")
    add_library(lz4_static STATIC ${LZ4_STATIC_SOURCES})
    target_include_directories(lz4_static PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/src/lz4/lib")
endif()
if(NOT TARGET lz4_static)
    message(FATAL_ERROR "Pinned LZ4 source did not provide a static lz4_static target")
endif()
add_subdirectory(src/brotli EXCLUDE_FROM_ALL)
# Abseil's standalone CMake project validates that a gtest target already
# exists. Register the exact googletest checkout before adding Abseil.
if(NOT TARGET gtest)
    add_subdirectory(src/googletest EXCLUDE_FROM_ALL)
endif()
add_subdirectory(src/abseil-cpp EXCLUDE_FROM_ALL)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/protobuf/CMakeLists.txt")
    # Newer AOSP protobuf revisions provide their own CMake project.
    add_subdirectory(src/protobuf EXCLUDE_FROM_ALL)
else()
    # API 33's pinned AOSP protobuf revision uses the legacy source layout.
    # Build that exact checkout with the project's maintained CMake source list.
    include(${CMAKE_CURRENT_SOURCE_DIR}/lib/libprotoc.cmake)
endif()
# Protobuf's CMake project may already have added this exact AOSP checkout.
if(NOT TARGET gtest)
    add_subdirectory(src/googletest EXCLUDE_FROM_ALL)
endif()
add_subdirectory(src/tinyxml2 EXCLUDE_FROM_ALL)

# building sdk-tools
add_subdirectory(lib)
add_subdirectory(build-tools)
add_subdirectory(platform-tools)
add_subdirectory(others)

set(CPACK_RESOURCE_FILE_LICENSE "${PROJECT_SOURCE_DIR}/LICENSE.txt")

set(CPACK_SOURCE_PACKAGE_FILE_NAME "${PROJECT_NAME}-${TOOLS_VERSION}")
set(CPACK_SOURCE_IGNORE_FILES 
    ${PROJECT_SOURCE_DIR}/.git
    ${PROJECT_SOURCE_DIR}/.gitignore
    ${PROJECT_SOURCE_DIR}/.gitmodules
    ${PROJECT_SOURCE_DIR}/get_source.py
    ${PROJECT_SOURCE_DIR}/build
    ${PROJECT_SOURCE_DIR}/patches
    ${PROJECT_SOURCE_DIR}/pack
    )
    
include(CPack)
